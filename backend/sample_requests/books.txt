# Replace {{baseUrl}}, {{username}}, {{password}}, {{adminToken}}, and {{bookId}}
# with Postman environment variables or concrete values before sending.

# Log in as an ADMIN user. Copy access_token from the response into adminToken.
curl --request POST '{{baseUrl}}/auth/login' \
  --header 'Content-Type: application/json' \
  --data '{
    "username": "{{username}}",
    "password": "{{password}}"
  }'

# List books
curl --request GET '{{baseUrl}}/books' \
  --header 'Authorization: Bearer {{adminToken}}'

# Search books and filter by inclusive publication-date bounds.
curl --get '{{baseUrl}}/books' \
  --data-urlencode 'q=le guin' \
  --data-urlencode 'date_from=1960-01-01T00:00:00Z' \
  --data-urlencode 'date_to=1970-12-31T23:59:59Z' \
  --header 'Authorization: Bearer {{adminToken}}'

# Get one book
curl --request GET '{{baseUrl}}/books/{{bookId}}' \
  --header 'Authorization: Bearer {{adminToken}}'

# Create a book with zero copies. The date must include a timezone offset.
curl --request POST '{{baseUrl}}/books' \
  --header 'Authorization: Bearer {{adminToken}}' \
  --header 'Content-Type: application/json' \
  --data '{
    "title": "The Left Hand of Darkness",
    "author": "Ursula K. Le Guin",
    "date": "1969-03-01T00:00:00-08:00",
    "isbn": "978-0-441-47812-5",
    "loan_duration_days": 14,
    "total_copies": 0
  }'

# Update one or more allowed fields.
curl --request PATCH '{{baseUrl}}/books/{{bookId}}' \
  --header 'Authorization: Bearer {{adminToken}}' \
  --header 'Content-Type: application/json' \
  --data '{
    "title": "The Dispossessed",
    "total_copies": 3
  }'

# Delete a book
curl --request DELETE '{{baseUrl}}/books/{{bookId}}' \
  --header 'Authorization: Bearer {{adminToken}}'

# Borrow, list, and return a loan as a USER. Define userToken and loanId.
curl --request POST '{{baseUrl}}/books/{{bookId}}/loans' \
  --header 'Authorization: Bearer {{userToken}}'

curl --request GET '{{baseUrl}}/loans/me' \
  --header 'Authorization: Bearer {{userToken}}'

curl --request POST '{{baseUrl}}/loans/{{loanId}}/return' \
  --header 'Authorization: Bearer {{userToken}}'

# List active loans for a book as an ADMIN.
curl --request GET '{{baseUrl}}/books/{{bookId}}/loans' \
  --header 'Authorization: Bearer {{adminToken}}'

# Reserve an unavailable title as a USER. Define reservationId.
curl --request POST '{{baseUrl}}/books/{{bookId}}/reservations' \
  --header 'Authorization: Bearer {{userToken}}'

curl --request GET '{{baseUrl}}/reservations/me' \
  --header 'Authorization: Bearer {{userToken}}'

# Confirm or cancel a ready reservation.
curl --request POST '{{baseUrl}}/reservations/{{reservationId}}/confirm' \
  --header 'Authorization: Bearer {{userToken}}'

curl --request DELETE '{{baseUrl}}/reservations/{{reservationId}}' \
  --header 'Authorization: Bearer {{userToken}}'

# Poll and mark notifications read.
curl --get '{{baseUrl}}/notifications' \
  --data-urlencode 'unread_only=true' \
  --data-urlencode 'limit=50' \
  --header 'Authorization: Bearer {{userToken}}'

curl --request PATCH '{{baseUrl}}/notifications/{{notificationId}}/read' \
  --header 'Authorization: Bearer {{userToken}}'
